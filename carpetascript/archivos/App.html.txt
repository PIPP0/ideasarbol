    function assignStackIndexes(ideas) {

      ideas.forEach(idea => {

          maxIndexPerNode[idea.nodeId] = next;
        }
      });

    // --- Persistencia en Google (Apps Script + Drive) ---
    // El servidor (Code.gs) guarda el estado en Drive y UNE los cambios de varias personas: cada guardado

    const BOOT = window.__BOOT || { params: {}, baseUrl: '', email: '', prefs: {} };
    const clone = (o) => JSON.parse(JSON.stringify(o));
    let cloudSynced = false;       // ya llegó el primer estado desde Google
    let cloudVersion = 0;
    let baseState = null;          // último estado conocido del servidor
    let savePending = false, saveInFlight = false, saveTimer = null, pollTimer = null;
    let votesTimer = null, lastVotesSent = '';

    function gsCall(fn, ...args) {
      return new Promise((resolve, reject) => {
        if (!(window.google && google.script && google.script.run)) { reject(new Error('Sin conexión con Google Apps Script')); return; }
        google.script.run.withSuccessHandler(resolve).withFailureHandler(reject)[fn](...args);
      });
    }

    // Los respaldos los crea el servidor (diario y antes de cualquier baja de ideas o árboles).
    function backupCloudState() { return Promise.resolve(); }

    function saveStateToFirestore() {
      if (!cloudSynced) return; // hasta conocer el estado del servidor no se guarda nada
      savePending = true;
      clearTimeout(saveTimer);
      saveTimer = setTimeout(flushCloudSave, 800);
    }

    async function flushCloudSave() {
      if (saveInFlight || !savePending || !cloudSynced) return;
      savePending = false;
      saveInFlight = true;
      const enviado = clone({ trees, appConfig });
      try {
        const r = await gsCall('apiSaveState', { base: baseState, next: enviado });
        const st = r && r.state;
        if (st) {
          baseState = clone({ trees: st.trees, appConfig: st.appConfig });
          cloudVersion = st.version;
          if (!savePending) applyRemoteState(st); // sin cambios nuevos en el camino: adopta lo unido
        }
      } catch (err) {
        console.error('Error guardando en Google (se reintenta):', err);
        savePending = true;
        showToast('No se pudo guardar en Google; reintentando…');
        clearTimeout(saveTimer);
        saveTimer = setTimeout(flushCloudSave, 6000);
      } finally {
        saveInFlight = false;
        if (savePending && !saveTimer) saveTimer = setTimeout(flushCloudSave, 300);
      }
    }

    function applyRemoteState(st) {
      if (!st || !Array.isArray(st.trees) || !st.trees.length) return;
      const nuevoCfg = st.appConfig || appConfig;
      if (JSON.stringify(trees) === JSON.stringify(st.trees) && JSON.stringify(appConfig) === JSON.stringify(nuevoCfg)) return;
      trees = clone(st.trees);
      normalizeAllTrees();
      if (!trees.some(t => t.id === currentTreeId)) currentTreeId = trees[0].id;
      if (st.appConfig) { appConfig = clone(st.appConfig); applyAppBackground(); }
      try { localStorage.setItem('arbol_ideas_multi_trees', JSON.stringify(trees)); } catch (e) {}
      applyCanvasTheme();
      updateTreePickerUI();
      renderSidebarCategories();
      renderTreeCards();
      renderConfigTreesList();
      // no redibujar el editor de categorías si la persona está escribiendo en él
      const ae = document.activeElement;
      if (!(ae && ae.closest && ae.closest('#categoriesList'))) renderConfigCategories();
      updateGlobalStats();
    }

    function startCloudPolling() {
      clearInterval(pollTimer);
      const revisar = async () => {
        if (document.hidden || !cloudSynced || saveInFlight || savePending) return;
        if (document.querySelector('#treeCardsContainer .is-dragging')) return;
        try {
          const v = await gsCall('apiGetVersion');
          if (v !== cloudVersion) {
            const st = await gsCall('apiGetState');
            if (savePending || saveInFlight) return;
            baseState = clone({ trees: st.trees, appConfig: st.appConfig });
            cloudVersion = st.version;
            applyRemoteState(st);
          }
        } catch (e) { /* se reintenta en el siguiente ciclo */ }
      };
      pollTimer = setInterval(revisar, 6000);
      document.addEventListener('visibilitychange', () => { if (!document.hidden) revisar(); });
    }

    function subscribeToFirestore() {
      gsCall('apiGetState').then(st => {
        cloudSynced = true;
        cloudVersion = st.version;
        baseState = clone({ trees: st.trees, appConfig: st.appConfig });
        if (st.userVotes) { userVotes = st.userVotes; lastVotesSent = JSON.stringify(userVotes); }
        applyRemoteState(st);
        updateGlobalStats();
        startCloudPolling();
      }).catch(err => {
        console.error('No se pudo conectar con Google Apps Script:', err);
        showToast('No se pudo conectar con Google. Se muestran los datos guardados en este navegador.');
      });
    }

    // Los votos (me gusta / es factible) de cada persona se guardan aparte, por correo.
    function scheduleVotesSave() {
      const s = JSON.stringify(userVotes);
      if (!cloudSynced || s === lastVotesSent) return;
      clearTimeout(votesTimer);
      votesTimer = setTimeout(() => {
        gsCall('apiGuardarVotos', clone(userVotes)).then(() => { lastVotesSent = s; }).catch(() => {});
      }, 600);
    }

    // Deja cada árbol con categorías/ideas/stackIndex consistentes, sin resucitar
    // árboles demo que el usuario haya eliminado a propósito (por eso NO se mezcla
    // aquí con DEMO_TREES; ese paso solo corre una vez, en el arranque local).
    function normalizeAllTrees() {
      trees.forEach(tree => {
        if (!tree.categories) {
          tree.categories = JSON.parse(JSON.stringify(DEFAULT_CATEGORIES));
        } else {
          Object.keys(DEFAULT_CATEGORIES).forEach(k => {
            if (!tree.categories[k]) {
              tree.categories[k] = JSON.parse(JSON.stringify(DEFAULT_CATEGORIES[k]));
            } else if (!tree.categories[k].desc) {
              tree.categories[k].desc = DEFAULT_CATEGORIES[k].desc;
            }
          });
        }
        if (!Array.isArray(tree.ideas)) {
          tree.ideas = [];
        }
        tree.ideas.forEach(normalizeIdeaFields);
        assignStackIndexes(tree.ideas);
      });
    }

    function saveState() {
      try {
        localStorage.setItem('arbol_ideas_multi_trees', JSON.stringify(trees));
        localStorage.setItem('arbol_ideas_current_tree_id', currentTreeId);
        localStorage.setItem('arbol_ideas_app_config', JSON.stringify(appConfig));
        localStorage.setItem('arbol_ideas_user_votes', JSON.stringify(userVotes));
      } catch (e) {
        console.error('Error al guardar datos:', e);
      }
      saveStateToFirestore();
      scheduleVotesSave();
      updateGlobalStats();
      updateUrlState();
    }

    function loadState() {
      try {
        const savedTrees = localStorage.getItem('arbol_ideas_multi_trees');
        if (savedTrees) {
          const parsed = JSON.parse(savedTrees);
          if (Array.isArray(parsed) && parsed.length > 0) {
            trees = parsed;
          }
        }

        DEMO_TREES.forEach(demo => {
          if (!trees.some(t => t.id === demo.id)) {
            trees.push(JSON.parse(JSON.stringify(demo)));
          }
        });

        normalizeAllTrees();

        const urlTreeId = getTreeIdFromUrl();
        if (urlTreeId && trees.some(t => t.id === urlTreeId)) {
          currentTreeId = urlTreeId;
        } else {
          const savedTreeId = localStorage.getItem('arbol_ideas_current_tree_id');
          if (savedTreeId && trees.some(t => t.id === savedTreeId)) {
            currentTreeId = savedTreeId;
          } else if (trees.length > 0) {
            currentTreeId = trees[0].id;
          }
        }

        const savedConfig = localStorage.getItem('arbol_ideas_app_config');
        if (savedConfig) {
          appConfig = JSON.parse(savedConfig);
        }

        const savedVotes = localStorage.getItem('arbol_ideas_user_votes');
        if (savedVotes) {
          userVotes = JSON.parse(savedVotes);
        }
      } catch (e) {
        console.error('Error al cargar estado:', e);
      }
      applyAppBackground();
      updateUrlState();
      subscribeToFirestore();
    }

    function getClosestBranchNode(x, y) {
      let closest = BRANCH_NODES[0];
      let minDistance = Infinity;

      BRANCH_NODES.forEach(node => {
        const dist = Math.hypot(node.x - x, node.y - y);
        if (dist < minDistance) {
          minDistance = dist;
          closest = node;
        }
      });

      return closest;
    }

    function getClosestCorner(cardLeft, cardTop, nodeX, nodeY, width = 220, height = 120) {
      const corners = [
        { x: cardLeft, y: cardTop, name: 'top-left' },
        { x: cardLeft + width, y: cardTop, name: 'top-right' },
        { x: cardLeft, y: cardTop + height, name: 'bottom-left' },
        { x: cardLeft + width, y: cardTop + height, name: 'bottom-right' }
      ];

      let bestCorner = corners[0];
      let minDist = Infinity;

      corners.forEach(corner => {
        const dist = Math.hypot(corner.x - nodeX, corner.y - nodeY);
        if (dist < minDist) {
          minDist = dist;
          bestCorner = corner;
        }
      });

      return bestCorner;
    }

    function screenToCanvasCoords(clientX, clientY) {
      const treeViewport = document.getElementById('treeViewport');
      const rect = treeViewport.getBoundingClientRect();
      const x = (clientX - rect.left) / zoomLevel;
      const y = (clientY - rect.top) / zoomLevel;
      return { x, y };
    }

    function updateDragTether(cardLeft, cardTop, targetNode, categoryColorHex = '#2dd4bf', width = 220, height = 120) {
      const tetherGroup = document.getElementById('activeDragTetherGroup');
      if (!tetherGroup) return;

      const closestCorner = getClosestCorner(cardLeft, cardTop, targetNode.x, targetNode.y, width, height);

      BRANCH_NODES.forEach(n => {
        const circleEl = document.getElementById(`node-${n.id}`);
        if (circleEl) {
          if (n.id === targetNode.id) {
            circleEl.classList.add('target-node-active');
            circleEl.setAttribute('r', '16');
          } else {
            circleEl.classList.remove('target-node-active');
            circleEl.setAttribute('r', '11');
          }
        }
      });

      tetherGroup.innerHTML = `
        <path d="M ${targetNode.x} ${targetNode.y} L ${closestCorner.x} ${closestCorner.y}"
              stroke="${categoryColorHex}" stroke-width="4" stroke-linecap="round" fill="none" class="active-tether" opacity="0.95" />
        <circle cx="${targetNode.x}" cy="${targetNode.y}" r="8" fill="${categoryColorHex}" filter="url(#glow)" />
        <circle cx="${closestCorner.x}" cy="${closestCorner.y}" r="7" fill="#ffffff" stroke="${categoryColorHex}" stroke-width="2.5" filter="url(#glow)" />
      `;
    }

    function setupDragAndDropHandlers() {
      const presetCards = document.querySelectorAll('.sidebar-preset-card');

      presetCards.forEach(card => {
        card.addEventListener('click', () => {
          if (activeDraggedCard) return;
          const categoryKey = card.getAttribute('data-type');
          const defaultNode = BRANCH_NODES[Math.floor(Math.random() * BRANCH_NODES.length)];
          openCreateModal(categoryKey, defaultNode.id, defaultNode.anchorCorner);
        });

        const handleStartDragPreset = (e) => {
          if (e.target.closest('button')) return;

          const categoryKey = card.getAttribute('data-type');
          const categories = getActiveCategories();
          const cat = categories[categoryKey] || Object.values(categories)[0];

          let targetNode = BRANCH_NODES[0];

          const onMove = (moveEvent) => {
            const moveClientX = moveEvent.touches ? moveEvent.touches[0].clientX : moveEvent.clientX;
            const moveClientY = moveEvent.touches ? moveEvent.touches[0].clientY : moveEvent.clientY;

            const coords = screenToCanvasCoords(moveClientX, moveClientY);
            targetNode = getClosestBranchNode(coords.x, coords.y);

            updateDragTether(coords.x - 110, coords.y - 60, targetNode, catHex(cat), 220, 120);
          };

          const onEnd = (endEvent) => {
            window.removeEventListener('mousemove', onMove);
            window.removeEventListener('mouseup', onEnd);
            window.removeEventListener('touchmove', onMove);
            window.removeEventListener('touchend', onEnd);

            clearDragTether();

            const endClientX = endEvent.changedTouches ? endEvent.changedTouches[0].clientX : endEvent.clientX;
            const endClientY = endEvent.changedTouches ? endEvent.changedTouches[0].clientY : endEvent.clientY;

            const canvasContainer = document.getElementById('canvasContainer');
            const rect = canvasContainer.getBoundingClientRect();

            if (endClientX >= rect.left && endClientX <= rect.right && endClientY >= rect.top && endClientY <= rect.bottom) {
              const coords = screenToCanvasCoords(endClientX, endClientY);
              const dropNode = getClosestBranchNode(coords.x, coords.y);
              const bestCorner = getClosestCorner(coords.x - 110, coords.y - 60, dropNode.x, dropNode.y, 220, 120);
              openCreateModal(categoryKey, dropNode.id, bestCorner.name);
            }
          };

          window.addEventListener('mousemove', onMove);
          window.addEventListener('mouseup', onEnd);
          window.addEventListener('touchmove', onMove, { passive: false });
          window.addEventListener('touchend', onEnd);
        };

        card.addEventListener('mousedown', handleStartDragPreset);
        card.addEventListener('touchstart', handleStartDragPreset, { passive: false });
      });
    }

    function renderTreeCards() {
      const activeTree = getActiveTree();
      const isLightTheme = !!window.DS_FORCE_LIGHT || activeTree.theme === 'light';
      const categories = getActiveCategories();
      const searchTerm = document.getElementById('searchInput').value.toLowerCase().trim();
      const catSelected = document.getElementById('categoryFilter').value;
      const treeCardsContainer = document.getElementById('treeCardsContainer');
      const connectorsGroup = document.getElementById('branchConnectors');
      const dynamicSubBranchesGroup = document.getElementById('dynamicSubBranches');

      treeCardsContainer.innerHTML = '';
      connectorsGroup.innerHTML = '';
      if (dynamicSubBranchesGroup) dynamicSubBranchesGroup.innerHTML = '';

      // Se asigna sobre TODAS las ideas del árbol (no solo las filtradas) para que la
      // posición de cada tarjeta sea estable sin importar el filtro/búsqueda activos.
      assignStackIndexes(activeTree.ideas);

      const filtered = activeTree.ideas.filter(idea => {
        const matchesCategory = (catSelected === 'all') || (idea.category === catSelected);
        const matchesSearch = idea.title.toLowerCase().includes(searchTerm) ||
                              idea.description.toLowerCase().includes(searchTerm) ||
                              idea.author.toLowerCase().includes(searchTerm);
        return matchesCategory && matchesSearch;
      });

      const nodeCounters = {};
      const cardDOMItems = [];

      filtered.forEach(idea => {
        let nodeKey = idea.nodeId;
        if (!nodeKey || !BRANCH_NODES.find(n => n.id === nodeKey)) {
          nodeKey = BRANCH_NODES[0].id;
          idea.nodeId = nodeKey;
        }
        nodeCounters[nodeKey] = (nodeCounters[nodeKey] || 0) + 1;
      });

      BRANCH_NODES.forEach(node => {
        const count = nodeCounters[node.id] || 0;
        const foliageEl = document.getElementById(`foliage-${node.id}`);
        if (foliageEl) {
          const baseRadius = 85;
          const grownRadius = baseRadius + (count * 28);
          foliageEl.setAttribute('r', grownRadius);
          foliageEl.setAttribute('opacity', count > 0 ? '0.85' : '0.35');
        }

        const circleEl = document.getElementById(`node-${node.id}`);
        if (circleEl) {
          circleEl.setAttribute('r', `${10 + Math.min(count * 3, 12)}`);
        }
      });

      filtered.forEach(idea => {
        normalizeIdeaFields(idea);
        const nodeKey = idea.nodeId;
        const node = BRANCH_NODES.find(n => n.id === nodeKey) || BRANCH_NODES[0];

        const stackIndex = idea.stackIndex || 0;

        const cat = categories[idea.category] || Object.values(categories)[0];
        const color = catHex(cat);
        const statusCfg = getStatusConfig(idea.status);
        const isCommentsExpanded = expandedCardComments.has(idea.id);
        const isDescExpanded = expandedCardDescriptions.has(idea.id);

        const cornerName = idea.anchorCorner || node.anchorCorner || 'bottom-right';

        // Calcular posición de la punta de la rama o sub-rama
        let tipX = node.x;

    function attachCardDragBehavior(cardEl, idea) {
      const categories = getActiveCategories();
      const cat = categories[idea.category] || Object.values(categories)[0];

      const handleStartDrag = (e) => {
        if (e.target.closest('button') || e.target.closest('input')) return;
        e.preventDefault();

        activeDraggedCard = idea.id;
        cardEl.classList.add('is-dragging');

        const isTouch = e.type === 'touchstart';
        const startClientX = isTouch ? e.touches[0].clientX : e.clientX;
        const startClientY = isTouch ? e.touches[0].clientY : e.clientY;

        const initialLeft = parseFloat(cardEl.style.left);
        const initialTop = parseFloat(cardEl.style.top);

        const cardWidth = cardEl.offsetWidth || 220;
        const cardHeight = cardEl.offsetHeight || 120;

        let targetNode = BRANCH_NODES.find(n => n.id === idea.nodeId) || BRANCH_NODES[0];

        const onMove = (moveEvent) => {
          const moveClientX = moveEvent.touches ? moveEvent.touches[0].clientX : moveEvent.clientX;
          const moveClientY = moveEvent.touches ? moveEvent.touches[0].clientY : moveEvent.clientY;

          const deltaX = (moveClientX - startClientX) / zoomLevel;
          const deltaY = (moveClientY - startClientY) / zoomLevel;

          const currentLeft = initialLeft + deltaX;
          const currentTop = initialTop + deltaY;

          cardEl.style.left = `${currentLeft}px`;
          cardEl.style.top = `${currentTop}px`;

          const cardCenterX = currentLeft + (cardWidth / 2);
          const cardCenterY = currentTop + (cardHeight / 2);

          targetNode = getClosestBranchNode(cardCenterX, cardCenterY);

          updateDragTether(currentLeft, currentTop, targetNode, catHex(cat), cardWidth, cardHeight);
        };

        const onEnd = () => {
          window.removeEventListener('mousemove', onMove);
          window.removeEventListener('mouseup', onEnd);
          window.removeEventListener('touchmove', onMove);
          window.removeEventListener('touchend', onEnd);

          cardEl.classList.remove('is-dragging');
          clearDragTether();
          activeDraggedCard = null;

          if (targetNode) {
            const finalLeft = parseFloat(cardEl.style.left);
            const finalTop = parseFloat(cardEl.style.top);

            const bestCorner = getClosestCorner(finalLeft, finalTop, targetNode.x, targetNode.y, cardWidth, cardHeight);

            if (idea.nodeId !== targetNode.id) {
              // Cambió de rama: libera su slot fijo anterior para que le asignen uno nuevo
              // en el nodo de destino (sin chocar con las tarjetas que ya están ahí).
              delete idea.stackIndex;
            }
            idea.nodeId = targetNode.id;
            idea.anchorCorner = bestCorner.name;

            saveState();

            const cornerLabels = {
              'top-left': 'superior izquierda',
              'top-right': 'superior derecha',
              'bottom-left': 'inferior izquierda',
              'bottom-right': 'inferior derecha'
            };
            showToast(`Conectada a "${targetNode.name}" (esquina ${cornerLabels[bestCorner.name]})`);
          }

          renderTreeCards();
        };

        window.addEventListener('mousemove', onMove);
        window.addEventListener('mouseup', onEnd);
        window.addEventListener('touchmove', onMove, { passive: false });
        window.addEventListener('touchend', onEnd);
      };

      cardEl.addEventListener('mousedown', handleStartDrag);
      cardEl.addEventListener('touchstart', handleStartDrag, { passive: false });
    }

    window.voteIdea = function(id, type, event) {
      if (event) event.stopPropagation();
      const activeTree = getActiveTree();
      const idea = activeTree.ideas.find(i => i.id === id);
      if (!idea) return;

      if (!userVotes[id]) {
        userVotes[id] = { hearts: false, stars: false };
      }

      const alreadyVoted = !!userVotes[id][type];

      if (alreadyVoted) {
        idea[type] = Math.max(0, (idea[type] || 0) - 1);
        userVotes[id][type] = false;
        showToast(type === 'hearts' ? 'Quitaste tu "Me gusta" ❤️' : 'Quitaste tu voto de "Es factible" ⭐');
      } else {
        idea[type] = (idea[type] || 0) + 1;
        userVotes[id][type] = true;
        showToast(type === 'hearts' ? '¡Te gusta esta idea! ❤️' : '¡Marcada como factible! ⭐');
      }

      const btn = event ? event.currentTarget : null;
      if (btn) {
        btn.classList.add('animate-pop');
        setTimeout(() => btn.classList.remove('animate-pop'), 250);
      }

      saveState();
      renderTreeCards();
    };

    // Traduce texto al español con el endpoint público y sin clave de Google Translate
    // (no es una API oficial/documentada, pero es gratuita y tiene CORS abierto).
    // Si falla o se limita, devuelve el texto original en vez de romper la búsqueda.
    async function translateToSpanish(text) {
      try {
        const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=es&dt=t&q=${encodeURIComponent(text)}`;
        const res = await fetch(url);
        if (!res.ok) throw new Error('HTTP ' + res.status);
        const data = await res.json();
        return data[0].map(chunk => chunk[0]).join('');
      } catch (err) {
        console.error('Error traduciendo:', err);
        return text;
      }
    }

    // El contenido fuente está en inglés, así que la búsqueda en español se traduce

    async function translateToEnglish(text) {

        const res = await fetch(url);

        return data[0].map(chunk => chunk[0]).join('');
      } catch (err) {
        console.error('Error traduciendo búsqueda:', err);
        return text;
      }
    }

    async function searchTrends(query) {
      const list = document.getElementById('trendsResultsList');
      const searchBtn = document.getElementById('trendsSearchBtn');
      const term = (query || '').trim();

      if (!term) {
        list.innerHTML = `<div class="text-center text-xs text-slate-500 py-8">Escribe una palabra clave para buscar tendencias.</div>`;
        return;
      }

      const requestId = ++trendsRequestId;
      searchBtn.disabled = true;
      list.innerHTML = `
        <div class="text-center text-xs text-slate-500 py-8 flex flex-col items-center gap-2">
          <i class="fa-solid fa-circle-notch fa-spin text-emerald-400"></i>
          Buscando tendencias del mes para "${escapeHtml(term)}"...
        </div>
      `;

      try {
        const searchTermEn = await translateToEnglish(term);
        if (requestId !== trendsRequestId) return;

        const startOfMonth = getStartOfCurrentMonthTimestamp();
        const url = `https://hn.algolia.com/api/v1/search?query=${encodeURIComponent(searchTermEn)}&tags=story&numericFilters=created_at_i%3E${startOfMonth}&hitsPerPage=12`;
        const response = await fetch(url);
        if (requestId !== trendsRequestId) return; // llegó una búsqueda más nueva mientras esperábamos
        if (!response.ok) throw new Error('HTTP ' + response.status);

        const data = await response.json();
        // "Ask HN" son preguntas de discusión, no tendencias/iniciativas concretas — se excluyen.
        const hits = (data.hits || []).filter(h => h.title && !/^Ask HN:/i.test(h.title));

        list.innerHTML = `
          <div class="text-center text-xs text-slate-500 py-8 flex flex-col items-center gap-2">
            <i class="fa-solid fa-circle-notch fa-spin text-emerald-400"></i>
            Traduciendo al español...
          </div>
        `;
        await Promise.all(hits.map(async h => {
          h.titleEs = await translateToSpanish(h.title);
        }));
        if (requestId !== trendsRequestId) return;

        trendsCache = {};
        hits.forEach(h => { trendsCache[h.objectID] = h; });

        renderTrendsList(hits, term);
      } catch (err) {
        if (requestId !== trendsRequestId) return;
        console.error('Error buscando tendencias:', err);
        list.innerHTML = `
          <div class="text-center text-xs text-rose-400 py-8 space-y-1.5">
            <i class="fa-solid fa-triangle-exclamation block text-base"></i>
            <p>No se pudo conectar con la API de tendencias.</p>
            <p class="text-slate-500">Revisa tu conexión e intenta de nuevo.</p>
          </div>
        `;
      } finally {
        if (requestId === trendsRequestId) searchBtn.disabled = false;
      }
    }

    function renderTrendsList(hits, term) {
      const list = document.getElementById('trendsResultsList');

      if (hits.length === 0) {
        list.innerHTML = `<div class="text-center text-xs text-slate-500 py-8">Sin resultados este mes para "${escapeHtml(term)}". Prueba con otra palabra clave o uno de los temas sugeridos.</div>`;
        return;
      }

      list.innerHTML = hits.map(hit => {
        let domain = '';
        try { domain = hit.url ? new URL(hit.url).hostname.replace('www.', '') : 'news.ycombinator.com'; } catch (e) { domain = ''; }
        const dateLabel = hit.created_at ? new Date(hit.created_at).toLocaleDateString('es-CL', { day: '2-digit', month: 'short', year: 'numeric' }) : '';

        return `
          <div class="glass-card p-3 rounded-xl border-l-2 border-amber-500/60 space-y-1.5">
            <a href="${escapeHtml(hit.url || `https://news.ycombinator.com/item?id=${hit.objectID}`)}" target="_blank" rel="noopener noreferrer" class="text-xs font-medium text-white hover:text-emerald-300 transition leading-snug line-clamp-2 block">
              ${escapeHtml(hit.titleEs || hit.title)}
            </a>
            <div class="flex items-center gap-2 text-[10px] text-slate-500">
              ${domain ? `<span class="truncate max-w-[110px]"><i class="fa-solid fa-link text-[9px] mr-1"></i>${escapeHtml(domain)}</span>` : ''}
              <span class="ml-auto flex items-center gap-1 shrink-0"><i class="fa-solid fa-arrow-up text-[9px]"></i>${hit.points ?? 0}</span>
              <span class="flex items-center gap-1 shrink-0"><i class="fa-solid fa-comment text-[9px]"></i>${hit.num_comments ?? 0}</span>
            </div>
            <div class="flex items-center justify-between pt-1">
              <span class="text-[9px] text-slate-600">${escapeHtml(dateLabel)}</span>
              <button onclick="convertTrendToIdea('${hit.objectID}')" type="button" class="px-2.5 py-1 text-[10px] font-semibold rounded-lg bg-emerald-500/20 text-emerald-300 border border-emerald-500/30 hover:bg-emerald-500/30 transition flex items-center gap-1">
                <i class="fa-solid fa-wand-magic-sparkles text-[9px]"></i> Convertir en idea
              </button>
            </div>
          </div>
        `;
      }).join('');
    }

    // Matriz de Priorización: Me gusta (impacto, eje Y) x Es factible (eje X)
    function renderPriorityMatrix() {
      const activeTree = getActiveTree();
      const categories = getActiveCategories();
      const container = document.getElementById('matrixDotsContainer');
      const legend = document.getElementById('matrixLegend');
      const emptyState = document.getElementById('matrixEmptyState');
      const subtitle = document.getElementById('matrixSubtitle');
      if (!container || !legend || !emptyState) return;

      const ideas = activeTree.ideas;
      subtitle.innerText = `Eje vertical: Me gusta (impacto) · Eje horizontal: Es factible · ${ideas.length} idea(s) en "${activeTree.name}"`;

      if (ideas.length === 0) {
        container.innerHTML = '';
        legend.innerHTML = '';
        emptyState.classList.remove('hidden');
        emptyState.classList.add('flex');
        return;
      }
      emptyState.classList.add('hidden');
      emptyState.classList.remove('flex');

      const maxHearts = Math.max(1, ...ideas.map(i => i.hearts || 0));
      const maxStars = Math.max(1, ...ideas.map(i => i.stars || 0));
      const maxVotes = Math.max(1, ...ideas.map(i => (i.hearts || 0) + (i.stars || 0)));

      container.innerHTML = ideas.map(idea => {
        normalizeIdeaFields(idea);
        const cat = categories[idea.category] || Object.values(categories)[0];
        const color = catHex(cat);
        const statusCfg = getStatusConfig(idea.status);

        const leftPct = 8 + ((idea.stars || 0) / maxStars) * 84;
        const topPct = 92 - ((idea.hearts || 0) / maxHearts) * 84;
        const totalVotes = (idea.hearts || 0) + (idea.stars || 0);
        const size = 22 + Math.round((totalVotes / maxVotes) * 20);

        return `
          <button type="button" onclick="editIdeaFromCard('${idea.id}')"
            class="absolute -translate-x-1/2 -translate-y-1/2 rounded-full border-2 flex items-center justify-center hover:scale-125 hover:z-10 transition-transform shadow-lg"
            style="left: ${leftPct}%; top: ${topPct}%; width: ${size}px; height: ${size}px; background-color: ${color}30; border-color: ${color};"
            title="${escapeHtml(idea.title)}&#10;${escapeHtml(statusCfg.label)} · Me gusta: ${idea.hearts} · Es factible: ${idea.stars}">
            <i class="fa-solid ${cat.icon || 'fa-lightbulb'} text-[9px]" style="color: ${dsText(color)};"></i>
          </button>
        `;
      }).join('');

      const usedCategoryIds = [...new Set(ideas.map(i => i.category))];
      legend.innerHTML = usedCategoryIds.map(catId => {
        const cat = categories[catId] || Object.values(categories)[0];
        const color = catHex(cat);
        return `
          <span class="text-[10px] flex items-center gap-1.5 text-slate-400">
            <span class="w-2.5 h-2.5 rounded-full" style="background-color: ${color};"></span>
            ${escapeHtml(cat.name)}
          </span>
        `;
      }).join('');
    }

    // Exportar ideas del árbol activo a CSV para llevar a comité de decisión
    function csvEscape(value) {
      const str = String(value === undefined || value === null ? '' : value);
      if (/[",\n]/.test(str)) {
        return `"${str.replace(/"/g, '""')}"`;
      }
      return str;
    }

    function exportIdeasToCsv() {
      const activeTree = getActiveTree();
      const categories = getActiveCategories();

      if (activeTree.ideas.length === 0) {
        showToast('No hay ideas para exportar en este árbol');
        return;
      }

      const headers = [
        'Título', 'Categoría', 'Estado', 'Responsable', 'Área de negocio impactada',
        'Usuarios/clientes afectados', 'Esfuerzo estimado', 'Autor', 'Área / Depto.',
        'Me gusta', 'Es factible', 'Total votos', 'Comentarios', 'Creado'
      ];

      const rows = activeTree.ideas.map(idea => {
        normalizeIdeaFields(idea);
        const cat = categories[idea.category] || Object.values(categories)[0];
        const statusCfg = getStatusConfig(idea.status);
        return [
          idea.title,
          cat.name,
          statusCfg.label,
          idea.owner,
          idea.businessArea,
          IMPACT_EFFORT_LABELS[idea.impact] || idea.impact,
          IMPACT_EFFORT_LABELS[idea.effort] || idea.effort,
          idea.author,
          idea.dept,
          idea.hearts,
          idea.stars,
          (idea.hearts || 0) + (idea.stars || 0),
          idea.comments.length,
          new Date(idea.createdAt).toLocaleDateString('es-CL')
        ].map(csvEscape).join(',');
      });

      const csvContent = '﻿' + [headers.map(csvEscape).join(','), ...rows].join('\r\n');
      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      const safeTreeName = activeTree.name
        .normalize('NFD').replace(/[̀-ͯ]/g, '')
        .toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
      link.href = url;
      link.download = `arbol-ideas-${safeTreeName || 'export'}.csv`;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
      showToast('CSV exportado para comité de decisión');
    }

    window.deleteCategory = function(catId) {
      const activeTree = getActiveTree();
      if (Object.keys(activeTree.categories).length <= 1) {
        showToast('Debe existir al menos una categoría en el árbol');
        return;
      }

      const catName = activeTree.categories[catId] ? activeTree.categories[catId].name : 'esta categoría';

      showConfirmModal(`¿Eliminar la categoría "${catName}"? Las ideas ya creadas con esta categoría no se eliminarán, pero quedarán sin categoría válida.`, () => {
        delete activeTree.categories[catId];
        saveState();
        renderConfigCategories();
        renderSidebarCategories();
        renderTreeCards();
        showToast('Categoría eliminada');
      });
    };

    window.resetCategoryToOriginalColor = function(catId) {

      activeTree.categories[catId].glowColor = `${origColor}55`;

      if (customSwitch) customSwitch.checked = true;

      saveState();
      renderSidebarCategories();
      renderTreeCards();
      renderConfigCategories();
      showToast('Color original restablecido');
    };

    const CATEGORY_DESC_MAX = 160;

    function readableOn(hex) {
      const m = /^#?([0-9a-f]{6})$/i.exec(String(hex || '').trim());
      if (!m) return '#fff';
      const n = parseInt(m[1], 16), r = n >> 16, g = (n >> 8) & 255, b = n & 255;
      return (0.299 * r + 0.587 * g + 0.114 * b) > 170 ? '#141B1F' : '#fff';
    }

    function setupZoomControls() {
      const zoomInBtn = document.getElementById('zoomInBtn');
      const zoomOutBtn = document.getElementById('zoomOutBtn');
      const zoomResetBtn = document.getElementById('zoomResetBtn');
      const zoomText = document.getElementById('zoomLevelText');
      const canvasContainer = document.getElementById('canvasContainer');
      const treeViewport = document.getElementById('treeViewport');
      let userTookControlOfZoom = false;
      let focusedCardId = null;   // tarjeta acercada con un toque
      let savedView = null;       // vista anterior, para volver con Esc

      // El porcentaje que se muestra es relativo a la pantalla: 100 % = árbol completo ajustado a la
      // pantalla (el estado inicial). El mínimo es 50 % de ese tamaño, así que el árbol siempre se ve
      // completo y nunca queda diminuto, en cualquier tamaño de ventana.
      const ZOOM_MAX = 2.5;
      const ZOOM_FLOOR = 0.5;
      const MIN_VS_FIT = 0.5;
      let zoomMin = 0.3;
      let fitRef = null; // zoom que deja el árbol completo ajustado a la pantalla (= 100 % en pantalla)
      const zoomSlider = document.getElementById('zoomSlider');
      const clampZoom = (z) => Math.max(zoomMin, Math.min(ZOOM_MAX, z));
      // El deslizador usa escala logarítmica: cada tramo se siente igual de "largo" a cualquier nivel
      const zoomToSlider = (z) => Math.max(0, Math.min(100, 100 * Math.log(z / zoomMin) / Math.log(ZOOM_MAX / zoomMin)));
      const sliderToZoom = (v) => zoomMin * Math.pow(ZOOM_MAX / zoomMin, v / 100);

      // Márgenes mínimos: arriba queda libre el aviso flotante; abajo/derecha están los controles de zoom
      const FIT_MARGIN = { x: 10, top: 48, bottom: 10 };

      // El árbol nunca puede salir del marco: si cabe en pantalla, se mantiene completo dentro;
      // si está ampliado más que la pantalla, no se puede arrastrar más allá de sus bordes.
      let boundsCache = null;
      function axisRange(size, start, end, m0, m1) {
        const base = size / 2 + zoomLevel * (start - 700); // (el origen de escala es el centro del lienzo base)
        const len = zoomLevel * (end - start);
        const a = m0 - base;
        const b = size - m1 - len - base;
        return [Math.min(a, b), Math.max(a, b)];
      }
      function clampPan() {
        const rect = canvasContainer.getBoundingClientRect();
        if (rect.width < 100 || rect.height < 100) return;
        const b = boundsCache || (boundsCache = getContentBounds());
        const [x0, x1] = axisRange(rect.width, b.minX, b.maxX, FIT_MARGIN.x, FIT_MARGIN.x);
        panX = Math.max(x0, Math.min(x1, panX));
        // vertical: el origen de escala es el centro vertical del lienzo base (450)
        const base = rect.height / 2 + zoomLevel * (b.minY - 450);
        const len = zoomLevel * (b.maxY - b.minY);
        const a = FIT_MARGIN.top - base;
        const c = rect.height - FIT_MARGIN.bottom - len - base;
        panY = Math.max(Math.min(a, c), Math.min(Math.max(a, c), panY));
      }

      function updateTransform(disableTransition = false) {
        clampPan();
        if (disableTransition) {
          treeViewport.style.transition = 'none';
        } else {
          treeViewport.style.transition = 'transform 0.2s cubic-bezier(0.22, 1, 0.36, 1)';
        }
        treeViewport.style.transform = `translate(${panX}px, ${panY}px) scale(${zoomLevel})`;
        if (zoomText) {
          zoomText.innerText = `${Math.round((zoomLevel / (fitRef || zoomLevel)) * 100)}%`;
        }
        if (zoomSlider) {
          zoomSlider.value = zoomToSlider(zoomLevel);
          zoomSlider.setAttribute('aria-valuetext', `${Math.round(zoomLevel * 100)}%`);
        }
      }

      // Zoom que mantiene fijo el punto (ex, ey), medido desde el centro del lienzo.
      // Con (0, 0) acerca/aleja hacia el centro de lo que se ve; con la posición del cursor,
      // el punto bajo el cursor no se mueve.
      function zoomAt(newZoom, ex, ey, animate) {
        const z = clampZoom(newZoom);
        if (z === zoomLevel) return;
        clearFocusVisual();
        const s = z / zoomLevel;
        panX = ex - (ex - panX) * s;
        panY = ey - (ey - panY) * s;
        zoomLevel = z;
        updateTransform(!animate);
      }
      function zoomFromCenter(factor) {
        userTookControlOfZoom = true;
        zoomAt(zoomLevel * factor, 0, 0, true);
      }
      function setZoom(newZoom) { zoomAt(newZoom, 0, 0, true); }

      // Límites reales del contenido: el árbol dibujado (ramas, copa y base) más todas las tarjetas,
      // incluso las que sobresalen. Se usa el área ocupada por el dibujo, no el lienzo completo
      // de 1400x900, para no dejar bordes vacíos alrededor.
      function getContentBounds() {
        let minX = 80, minY = 100, maxX = 1320, maxY = 915;
        document.querySelectorAll('#treeCardsContainer > *').forEach(el => {
          const l = el.offsetLeft, t = el.offsetTop;
          const w = el.offsetWidth, h = el.offsetHeight;
          if (!w || !h) return;
          minX = Math.min(minX, l); minY = Math.min(minY, t);
          maxX = Math.max(maxX, l + w); maxY = Math.max(maxY, t + h);
        });
        return { minX, minY, maxX, maxY };
      }

      function computeFit() {
        const rect = canvasContainer.getBoundingClientRect();
        const b = getContentBounds();
        const bw = b.maxX - b.minX, bh = b.maxY - b.minY;
        const availW = rect.width - FIT_MARGIN.x * 2;
        const availH = rect.height - FIT_MARGIN.top - FIT_MARGIN.bottom;
        const z = Math.max(0.1, Math.min(1.5, Math.min(availW / bw, availH / bh)));
        return { z, b, rect, availH };
      }
      // Recalcula el mínimo permitido (ajuste a pantalla × 50 %)
      function refreshZoomMin() {
        const f = computeFit();
        boundsCache = f.b;
        fitRef = f.z;
        zoomMin = Math.max(0.05, f.z * MIN_VS_FIT); // 50 % del tamaño ajustado
        return f;
      }
      // Tras cambios de tamaño o de tarjetas: recalcula límites, respeta el mínimo y devuelve el árbol al marco
      function enforceView() {
        refreshZoomMin();
        if (zoomLevel < zoomMin) zoomLevel = zoomMin;
        updateTransform(true);
      }

      // Ajusta el zoom y centra el desplazamiento para que TODO el árbol (dibujo y
      // tarjetas) quepa en el área visible. Es el estado por defecto al cargar, al
      // cambiar de árbol y lo que hace el botón de "Ver todo el árbol".
      let lastFitZoom = null;
      // Centra el contenido (dibujo + tarjetas) en el área disponible para un zoom dado
      function placeContent(f, z) {
        const b = f.b;
        const bcx = (b.minX + b.maxX) / 2, bcy = (b.minY + b.maxY) / 2;
        const targetDy = (FIT_MARGIN.top + f.availH / 2) - f.rect.height / 2;
        panX = -z * (bcx - 700);
        panY = targetDy - z * (bcy - 450);
      }
      function fitTreeToContainer() {
        clearFocusVisual();
        const f = refreshZoomMin();
        lastFitZoom = f.z;
        zoomLevel = f.z;
        placeContent(f, zoomLevel);
        updateTransform(false);
      }

      // Ajuste automático: activo al cargar y al cambiar de árbol, mientras la persona
      // no tome el control del zoom/desplazamiento. Se reaplica cuando se vuelven a
      // dibujar las tarjetas (fuentes, datos de la nube) dentro de la ventana de tiempo.
      let autoFitUntil = Date.now() + 8000;
      window.requestAutoFit = function () {
        userTookControlOfZoom = false;
        autoFitUntil = Date.now() + 4000;
        fitTreeToContainer();
      };
      // Tocar o arrastrar una tarjeta también cuenta como tomar el control: la vista no se mueve sola.
      canvasContainer.addEventListener('mousedown', (e) => {
        if (e.target.closest('.glass-card')) userTookControlOfZoom = true;
      }, true);
      window.autoFitIfRequested = function () {
        if (window.reapplyCardFocus) window.reapplyCardFocus();
        if (!userTookControlOfZoom && Date.now() < autoFitUntil) fitTreeToContainer();
        else enforceView();
      };

      // Pasos proporcionales (+25 % / −20 %): se sienten parejos y son fáciles de revertir
      if (zoomInBtn) {
        zoomInBtn.onclick = (e) => { e.stopPropagation(); zoomFromCenter(1.25); };
      }
      if (zoomOutBtn) {
        zoomOutBtn.onclick = (e) => { e.stopPropagation(); zoomFromCenter(1 / 1.25); };
      }
      if (zoomSlider) {
        zoomSlider.addEventListener('input', (e) => {
          e.stopPropagation();
          userTookControlOfZoom = true;
          zoomAt(sliderToZoom(parseFloat(zoomSlider.value)), 0, 0, false);
        });
        zoomSlider.addEventListener('mousedown', (e) => e.stopPropagation());
      }
      // ---------- Acercar al tocar una tarjeta ----------
      const focusBackBtn = document.getElementById('focusBackBtn');
      function applyFocusVisual() {
        const box = document.getElementById('treeCardsContainer');
        if (!box) return;
        box.classList.toggle('tree-has-focus', !!focusedCardId);
        box.querySelectorAll('.card-focused').forEach(el => el.classList.remove('card-focused'));
        if (focusedCardId) {
          const el = box.querySelector(`[data-id="${focusedCardId}"]`);
          if (el) el.classList.add('card-focused');
        }
        if (focusBackBtn) focusBackBtn.classList.toggle('hidden', !focusedCardId);
      }
      // Quita el atenuado y olvida la vista anterior (la vista actual se conserva)
      function clearFocusVisual() {
        if (!focusedCardId && !savedView) return;
        focusedCardId = null;
        savedView = null;
        applyFocusVisual();
      }
      function focusCard(cardEl) {
        const id = cardEl.getAttribute('data-id');
        if (!id) return;
        if (focusedCardId === id) return;
        if (!focusedCardId) {
          savedView = { zoom: zoomLevel, panX, panY, userTook: userTookControlOfZoom };
        }
        focusedCardId = id;
        userTookControlOfZoom = true;
        const rect = canvasContainer.getBoundingClientRect();
        const w = cardEl.offsetWidth, h = cardEl.offsetHeight;
        const cx = cardEl.offsetLeft + w / 2, cy = cardEl.offsetTop + h / 2;
        // zoom cómodo para leer (≈130 %), sin que la tarjeta pase del 75 % del área visible
        const comfy = Math.min(1.3, (rect.width * 0.75) / w, (rect.height * 0.75) / h);
        zoomLevel = clampZoom(Math.max(zoomLevel, comfy));
        panX = -zoomLevel * (cx - 700);
        panY = -zoomLevel * (cy - 450);
        updateTransform(false);
        applyFocusVisual();
      }
      function exitFocus() {
        if (!focusedCardId || !savedView) { clearFocusVisual(); return; }
        const v = savedView;
        focusedCardId = null;
        savedView = null;
        zoomLevel = clampZoom(v.zoom);
        panX = v.panX; panY = v.panY;
        userTookControlOfZoom = v.userTook;
        applyFocusVisual();
        updateTransform(false);
      }
      window.reapplyCardFocus = applyFocusVisual; // tras redibujar las tarjetas
      if (focusBackBtn) focusBackBtn.addEventListener('click', (e) => { e.stopPropagation(); exitFocus(); });

      // Distinguir un toque de un arrastre: se acerca solo si el puntero casi no se movió
      let tapStart = null;
      canvasContainer.addEventListener('pointerdown', (e) => {
        if (e.button !== undefined && e.button !== 0) { tapStart = null; return; }
        const onUi = e.target.closest && e.target.closest('#zoomControlsPanel, #focusBackBtn, #dragHelpNotice');
        if (onUi) { tapStart = null; return; }
        const card = e.target.closest && e.target.closest('#treeCardsContainer .glass-card, #treeCardsContainer .glass-card-light');
        const interactive = e.target.closest && e.target.closest('button, a, input, textarea, select, label');
        tapStart = { x: e.clientX, y: e.clientY, card: (card && !interactive) ? card : null, bg: !card };
      }, true);
      canvasContainer.addEventListener('pointerup', (e) => {
        const t = tapStart; tapStart = null;
        if (!t) return;
        const moved = Math.hypot(e.clientX - t.x, e.clientY - t.y);
        if (t.card && moved < 6) { focusCard(t.card); }
        else if (t.bg && focusedCardId) {
          if (moved < 6) exitFocus(); else clearFocusVisual(); // arrastró el lienzo: conserva la vista
        }
      }, true);
      canvasContainer.addEventListener('pointercancel', () => { tapStart = null; }, true);

      // Atajos de teclado: + acerca, − aleja, 0 ajusta todo el árbol
      document.addEventListener('keydown', (e) => {
        if (e.ctrlKey || e.metaKey || e.altKey) return;
        const t = e.target;
        if (t && (/^(INPUT|TEXTAREA|SELECT)$/.test(t.tagName) || t.isContentEditable)) return;
        if (['ideaModal', 'configModal', 'confirmDeleteModal'].some(id => { const m = document.getElementById(id); return m && !m.classList.contains('hidden'); })) return;
        if (e.key === 'Escape' && focusedCardId) { e.preventDefault(); exitFocus(); }
        else if (e.key === '+' || e.key === '=') { e.preventDefault(); zoomFromCenter(1.25); }
        else if (e.key === '-' || e.key === '_') { e.preventDefault(); zoomFromCenter(1 / 1.25); }
        else if (e.key === '0') { e.preventDefault(); window.requestAutoFit(); }
      });

      if (zoomResetBtn) {
        zoomResetBtn.onclick = (e) => {
          e.stopPropagation();
          try {
            // vuelve al estado inicial: árbol completo ajustado a la pantalla y adaptándose a ella
            window.requestAutoFit();
          } catch (err) {
            console.warn('Restablecer vista: ajuste de respaldo', err);
            userTookControlOfZoom = true;
            fitTreeToContainer();
          }
        };
      }

      // Zoom con rueda / trackpad centrado en el cursor (el punto bajo el cursor no se mueve).
      //  - Rueda de mouse (saltos de ~100): un paso fijo y suave de 15 % por "clic".
      //  - Trackpad / pellizco (valores pequeños y continuos): proporcional a lo que se mueve el dedo,
      //    acumulado por fotograma y con tope, para que no se dispare con la inercia.
      let wheelAcc = 0;
      let wheelRaf = null;
      let wheelPoint = { ex: 0, ey: 0 };
      canvasContainer.addEventListener('wheel', (e) => {
        if (e.target.closest && e.target.closest('#zoomControlsPanel')) return;
        e.preventDefault();
        userTookControlOfZoom = true;

        let dy = e.deltaY;
        if (e.deltaMode === 1) dy *= 16; else if (e.deltaMode === 2) dy *= 100;
        const cr = canvasContainer.getBoundingClientRect();
        const ex = e.clientX - (cr.left + cr.width / 2);
        const ey = e.clientY - (cr.top + cr.height / 2);

        const isNotch = !e.ctrlKey && Math.abs(dy) >= 50 && Number.isInteger(e.deltaY);
        if (isNotch) {
          zoomAt(zoomLevel * (dy < 0 ? 1.15 : 1 / 1.15), ex, ey, true);
          return;
        }
        wheelAcc += -dy * (e.ctrlKey ? 0.010 : 0.0035);
        wheelPoint = { ex, ey };
        if (!wheelRaf) {
          wheelRaf = requestAnimationFrame(() => {
            wheelRaf = null;
            const step = Math.max(-0.06, Math.min(0.06, wheelAcc)); // tope por fotograma
            wheelAcc = 0;
            if (step) zoomAt(zoomLevel * Math.exp(step), wheelPoint.ex, wheelPoint.ey, false);
          });
        }
      }, { passive: false });

      let isPanning = false;
      let startX = 0;
      let startY = 0;

      canvasContainer.addEventListener('mousedown', (e) => {
        if (e.target.closest('.glass-card') || e.target.closest('button') || e.target.closest('input') || e.target.closest('select')) return;
        userTookControlOfZoom = true;
        isPanning = true;
        startX = e.clientX - panX;
        startY = e.clientY - panY;
        canvasContainer.style.cursor = 'grabbing';
      });

      window.addEventListener('mousemove', (e) => {
        if (!isPanning) return;
        panX = e.clientX - startX;
        panY = e.clientY - startY;
        updateTransform(true);
      });

      window.addEventListener('mouseup', () => {
        if (isPanning) {
          isPanning = false;
          canvasContainer.style.cursor = 'grab';
        }
      });

      let touchStartDist = null;

      canvasContainer.addEventListener('touchstart', (e) => {
        userTookControlOfZoom = true;
        if (e.touches.length === 1) {
          if (e.target.closest('.glass-card') || e.target.closest('button')) return;
          isPanning = true;
          startX = e.touches[0].clientX - panX;
          startY = e.touches[0].clientY - panY;
        } else if (e.touches.length === 2) {
          isPanning = false;
          touchStartDist = Math.hypot(
            e.touches[0].clientX - e.touches[1].clientX,
            e.touches[0].clientY - e.touches[1].clientY
          );
        }
      }, { passive: true });

      canvasContainer.addEventListener('touchmove', (e) => {
        if (isPanning && e.touches.length === 1) {
          panX = e.touches[0].clientX - startX;
          panY = e.touches[0].clientY - startY;
          updateTransform(true);
        } else if (e.touches.length === 2 && touchStartDist) {
          const currentDist = Math.hypot(
            e.touches[0].clientX - e.touches[1].clientX,
            e.touches[0].clientY - e.touches[1].clientY
          );
          const factor = currentDist / touchStartDist;
          zoomLevel = clampZoom(zoomLevel * factor);
          updateTransform(true);
          touchStartDist = currentDist;
        }
      }, { passive: true });

      canvasContainer.addEventListener('touchend', () => {
        isPanning = false;
        touchStartDist = null;
      });

      // Se usa ResizeObserver (en vez de adivinar con rAF/timeouts) para detectar
      // el momento real en que el contenedor alcanza su tamaño final de layout —
      // al cargar, el CSS de Tailwind (inyectado vía CDN) puede tardar en aplicarse
      // y el contenedor pasar por anchos intermedios antes de llegar a su tamaño
      // final — en vez de adivinar cuándo terminó, se recalcula en cada cambio de
      // tamaño hasta que el usuario interactúa manualmente con el zoom/pan
      // (userTookControlOfZoom), momento en que se deja de pisar su ajuste.
      const initialFitObserver = new ResizeObserver(() => {
        if (userTookControlOfZoom) {
          const rect0 = canvasContainer.getBoundingClientRect();
          if (rect0.width > 100 && rect0.height > 100) {
            const prev = lastFitZoom;
            const prevMin = zoomMin;
            const wasAtMin = zoomLevel <= prevMin * 1.03;
            const f = refreshZoomMin();
            if (wasAtMin) {
              // estaba en el mínimo: sigue en el mínimo de la pantalla nueva (árbol completo a la vista)
              zoomLevel = zoomMin;
              placeContent(f, zoomLevel);
              lastFitZoom = f.z;
              updateTransform(true);
              return;
            }
            if (prev && f.z !== prev) {
              if (Math.abs(zoomLevel / prev - 1) < 0.03) {
                fitTreeToContainer(); // estaba viendo el árbol completo: sigue viéndolo completo
                return;
              }
              // si había hecho zoom, mantiene la misma proporción respecto al ajuste de la pantalla nueva
              const k = f.z / prev;
              zoomLevel = clampZoom(zoomLevel * k);
              panX *= k; panY *= k;
            }
            lastFitZoom = f.z;
          }
          enforceView();
          return;
        }
        const rect = canvasContainer.getBoundingClientRect();
        if (rect.width > 100 && rect.height > 100) {
          fitTreeToContainer();
        }
      });
      initialFitObserver.observe(canvasContainer);
    }

      const o = opts || {};
      const modal = document.getElementById('confirmDeleteModal');
      document.getElementById('confirmDeleteMessage').innerText = message;
      const titleEl = modal.querySelector('h3');
      if (titleEl) titleEl.textContent = o.title || 'Confirmar eliminación';
      const okBtn = document.getElementById('confirmDeleteConfirmBtn');
      const okLabel = okBtn.querySelector('span');
      const okIcon = okBtn.querySelector('i');

      if (okIcon) okIcon.className = 'fa-solid ' + (o.icon || 'fa-trash-can') + ' text-xs';
      pendingDeleteAction = onConfirm;
      modal.classList.remove('hidden');
    }

    function escapeHtml(str) {
      if (!str) return '';
      return str.replace(/[&<>"']/g, function(m) {
        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
      });
    }
